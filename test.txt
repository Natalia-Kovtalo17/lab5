# GET: картинки немає в кеші (очікується 200, X-Cache: MISS)
curl -s -D - -o cat200.jpg http://127.0.0.1:3000/200

# GET: та сама картинка повторно (очікується 200, X-Cache: HIT)
curl -s -D - -o cat200.jpg http://127.0.0.1:3000/200

# GET: некоректний шлях (очікується 404)
curl -i http://127.0.0.1:3000/abc
# PUT: записати власну картинку в кеш (очікується 201)
curl -i -X PUT --data-binary @mypic.jpg http://127.0.0.1:3000/200

# GET після PUT: повертається картинка з PUT, а не з http.cat (очікується 200, X-Cache: HIT, Content-Length 31133)
curl -s -D - -o out.jpg http://127.0.0.1:3000/200

# PUT з порожнім тілом (очікується 400)
curl -i -X PUT http://127.0.0.1:3000/200
